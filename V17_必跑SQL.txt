alter table employees add column if not exists position_allowance numeric default 0;
alter table employees add column if not exists fixed_fuel_allowance numeric default 0;

create or replace function peihong_labor_insured_salary(p_wage numeric)
returns numeric
language plpgsql
immutable
as $$
begin
  if coalesce(p_wage,0) >= 43901 then return 45800;
  elsif coalesce(p_wage,0) >= 42001 then return 43900;
  elsif coalesce(p_wage,0) >= 40101 then return 42000;
  elsif coalesce(p_wage,0) >= 38201 then return 40100;
  elsif coalesce(p_wage,0) >= 36301 then return 38200;
  elsif coalesce(p_wage,0) >= 34801 then return 36300;
  elsif coalesce(p_wage,0) >= 33301 then return 34800;
  elsif coalesce(p_wage,0) >= 31801 then return 33300;
  elsif coalesce(p_wage,0) >= 30301 then return 31800;
  elsif coalesce(p_wage,0) >= 29501 then return 30300;
  else return 29500;
  end if;
end;
$$;

create or replace function peihong_health_insured_salary(p_wage numeric)
returns numeric
language plpgsql
immutable
as $$
begin
  if coalesce(p_wage,0) >= 101101 then return 105600;
  elsif coalesce(p_wage,0) >= 96601 then return 101100;
  elsif coalesce(p_wage,0) >= 92101 then return 96600;
  elsif coalesce(p_wage,0) >= 87601 then return 92100;
  elsif coalesce(p_wage,0) >= 83901 then return 87600;
  elsif coalesce(p_wage,0) >= 80201 then return 83900;
  elsif coalesce(p_wage,0) >= 76501 then return 80200;
  elsif coalesce(p_wage,0) >= 72801 then return 76500;
  elsif coalesce(p_wage,0) >= 69801 then return 72800;
  elsif coalesce(p_wage,0) >= 66801 then return 69800;
  elsif coalesce(p_wage,0) >= 63801 then return 66800;
  elsif coalesce(p_wage,0) >= 60801 then return 63800;
  elsif coalesce(p_wage,0) >= 57801 then return 60800;
  elsif coalesce(p_wage,0) >= 55401 then return 57800;
  elsif coalesce(p_wage,0) >= 53001 then return 55400;
  elsif coalesce(p_wage,0) >= 50601 then return 53000;
  elsif coalesce(p_wage,0) >= 48201 then return 50600;
  elsif coalesce(p_wage,0) >= 45801 then return 48200;
  elsif coalesce(p_wage,0) >= 43901 then return 45800;
  elsif coalesce(p_wage,0) >= 42001 then return 43900;

create or replace function peihong_sync_employee_insurance()
returns trigger
language plpgsql
as $$
declare
  wage_base numeric;
  dep_count numeric;
  labor_grade numeric;
  health_grade numeric;
begin
  wage_base := coalesce(new.salary,0)
    + coalesce(new.meal_allowance,0)
    + coalesce(new.position_allowance,0)
    + coalesce(new.fixed_fuel_allowance,0)
    + coalesce(new.fixed_performance_bonus,0)
    + coalesce(new.other_fixed_allowance,0);

  dep_count := least(greatest(coalesce(new.health_dependents_count,0),0),3);
  labor_grade := peihong_labor_insured_salary(wage_base);
  health_grade := peihong_health_insured_salary(wage_base);

  new.labor_insured_salary := labor_grade;
  new.health_insured_salary := health_grade;
  new.labor_insurance := round(labor_grade * 0.125 * 0.2);
  new.health_insurance := round(health_grade * 0.0517 * 0.3);
  new.dependent_health_insurance := round(health_grade * 0.0517 * 0.3 * dep_count);
  new.health_dependents_count := dep_count;

  return new;
end;
$$;

drop trigger if exists trg_peihong_sync_employee_insurance on employees;
create trigger trg_peihong_sync_employee_insurance
before insert or update of salary, meal_allowance, position_allowance, fixed_fuel_allowance, fixed_performance_bonus, other_fixed_allowance, health_dependents_count
on employees
for each row
execute function peihong_sync_employee_insurance();

-- 立即重算現有員工資料
update employees
set salary = coalesce(salary,0),
    meal_allowance = coalesce(meal_allowance,0),
    position_allowance = coalesce(position_allowance,0),
    fixed_fuel_allowance = coalesce(fixed_fuel_allowance,0),
    fixed_performance_bonus = coalesce(fixed_performance_bonus,0),
    other_fixed_allowance = coalesce(other_fixed_allowance,0),
    health_dependents_count = coalesce(health_dependents_count,0);

select
  employee_no,
  name,
  salary,
  meal_allowance,
  position_allowance,
  fixed_fuel_allowance,
  fixed_performance_bonus,
  other_fixed_allowance,
  (coalesce(salary,0)+coalesce(meal_allowance,0)+coalesce(position_allowance,0)+coalesce(fixed_fuel_allowance,0)+coalesce(fixed_performance_bonus,0)+coalesce(other_fixed_allowance,0)) as wage_base_total,
  labor_insured_salary,
  health_insured_salary,
  labor_insurance,
  health_insurance,
  dependent_health_insurance
from employees
order by employee_no;
