update employees
set
  login_account = coalesce(nullif(login_account, ''), employee_no),
  meal_allowance = coalesce(meal_allowance, 0),
  position_allowance = coalesce(position_allowance, 0),
  fixed_fuel_allowance = coalesce(fixed_fuel_allowance, 0),
  fixed_performance_bonus = coalesce(fixed_performance_bonus, 0),
  other_fixed_allowance = coalesce(other_fixed_allowance, 0),
  health_dependents_count = coalesce(health_dependents_count, 0),
  labor_insured_salary = coalesce(labor_insured_salary, 0),
  health_insured_salary = coalesce(health_insured_salary, 0),
  is_active = coalesce(is_active, true);

create index if not exists employees_login_account_idx on employees(login_account);
create index if not exists employees_employee_no_idx on employees(employee_no);

-- 2) 加班申請：補上 Excel 加班類型
create table if not exists overtime_requests (
  id uuid primary key default uuid_generate_v4(),
  employee_id uuid references employees(id) on delete cascade,
  overtime_date date not null,
  start_time time not null,
  end_time time not null,
  overtime_hours numeric default 0,
  overtime_type text default 'weekday_first2',
  reason text,
  status text default 'pending',
  created_at timestamp default now()
);
alter table overtime_requests add column if not exists overtime_type text default 'weekday_first2';
alter table overtime_requests add column if not exists reviewed_at timestamp;
alter table overtime_requests add column if not exists reviewed_by text;

-- 5) 安全登入 RPC：員工登入不再需要前端直接讀全部密碼
create or replace function employee_login(p_account text, p_password text)
returns setof employees
language sql
security definer
set search_path = public
as $$
  select *
  from employees
  where is_active is not false
    and (login_account = p_account or employee_no = p_account)
    and coalesce(login_password, '') = p_password
  limit 1;
$$;

grant execute on function employee_login(text, text) to anon, authenticated;

-- 注意：目前此系統是純 Netlify 前端 + Supabase publishable key，沒有真正的 Supabase Auth 使用者。
-- 因此前端 HR 後台仍需讀寫資料；這是「可運作的明確政策」，比直接 disable RLS 好管理，但若要正式高安全上線，建議再升級成 Supabase Auth + Edge Function。

alter table employees enable row level security;
alter table attendance enable row level security;
alter table leave_requests enable row level security;
alter table overtime_requests enable row level security;
alter table payroll enable row level security;

-- employees policies
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE schemaname='public' AND tablename='employees' AND policyname='peihong_v16_employees_select') THEN
    CREATE POLICY peihong_v16_employees_select ON employees FOR SELECT TO anon, authenticated USING (true);
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE schemaname='public' AND tablename='employees' AND policyname='peihong_v16_employees_insert') THEN
    CREATE POLICY peihong_v16_employees_insert ON employees FOR INSERT TO anon, authenticated WITH CHECK (true);
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE schemaname='public' AND tablename='employees' AND policyname='peihong_v16_employees_update') THEN
    CREATE POLICY peihong_v16_employees_update ON employees FOR UPDATE TO anon, authenticated USING (true) WITH CHECK (true);
  END IF;
END $$;
