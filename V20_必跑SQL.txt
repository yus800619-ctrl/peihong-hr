create or replace function peihong_labor_grade_v20(p_wage numeric)
returns numeric language plpgsql immutable as $$
begin
  if coalesce(p_wage,0) >= 43901 then return 45800;
  elsif coalesce(p_wage,0) >= 42001 then return 43900;
  elsif coalesce(p_wage,0) >= 40101 then return 42000;
  elsif coalesce(p_wage,0) >= 38201 then return 40100;
  elsif coalesce(p_wage,0) >= 36301 then return 38200;
  elsif coalesce(p_wage,0) >= 34801 then return 36300;
  elsif coalesce(p_wage,0) >= 33301 then return 34800;
  elsif coalesce(p_wage,0) >= 31801 then return 33300;
  elsif coalesce(p_wage,0) >= 30301 then return 31800;
  elsif coalesce(p_wage,0) >= 29501 then return 30300;
  else return 29500;
  end if;
end; $$;

create or replace function peihong_health_grade_v20(p_wage numeric)

begin

  elsif coalesce(p_wage,0) >= 96601 then return 101100;
  elsif coalesce(p_wage,0) >= 92101 then return 96600;
  elsif coalesce(p_wage,0) >= 87601 then return 92100;
  elsif coalesce(p_wage,0) >= 83901 then return 87600;
  elsif coalesce(p_wage,0) >= 80201 then return 83900;
  elsif coalesce(p_wage,0) >= 76501 then return 80200;
  elsif coalesce(p_wage,0) >= 72801 then return 76500;
  elsif coalesce(p_wage,0) >= 69801 then return 72800;
  elsif coalesce(p_wage,0) >= 66801 then return 69800;
  elsif coalesce(p_wage,0) >= 63801 then return 66800;
  elsif coalesce(p_wage,0) >= 60801 then return 63800;
  elsif coalesce(p_wage,0) >= 57801 then return 60800;
  elsif coalesce(p_wage,0) >= 55401 then return 57800;
  elsif coalesce(p_wage,0) >= 53001 then return 55400;
  elsif coalesce(p_wage,0) >= 50601 then return 53000;
  elsif coalesce(p_wage,0) >= 48201 then return 50600;
  elsif coalesce(p_wage,0) >= 45801 then return 48200;
  elsif coalesce(p_wage,0) >= 43901 then return 45800;
  elsif coalesce(p_wage,0) >= 42001 then return 43900;

create or replace function peihong_wage_base_v20(
  p_salary numeric, p_meal numeric, p_position numeric, p_fuel numeric, p_bonus numeric, p_other numeric
) returns numeric language sql immutable as $$
  select coalesce(p_salary,0)+coalesce(p_meal,0)+coalesce(p_position,0)+coalesce(p_fuel,0)+coalesce(p_bonus,0)+coalesce(p_other,0);
$$;

create or replace function peihong_annual_leave_days_v20(p_hire_date date, p_as_of date default current_date)
returns numeric language plpgsql immutable as $$
declare
  months integer;
  years integer;
begin
  if p_hire_date is null then return 0; end if;
  months := (date_part('year', age(p_as_of, p_hire_date))::int * 12) + date_part('month', age(p_as_of, p_hire_date))::int;
  years := date_part('year', age(p_as_of, p_hire_date))::int;
  if months < 6 then return 0;
  elsif months < 12 then return 3;
  elsif years < 2 then return 7;
  elsif years < 3 then return 10;
  elsif years < 5 then return 14;
  elsif years < 10 then return 15;
  else return least(30, 15 + (years - 10 + 1));
  end if;
end; $$;

create or replace function peihong_sync_employee_v20()
returns trigger language plpgsql as $$
declare
  wage_base numeric;
  dep_count numeric;
begin
  wage_base := peihong_wage_base_v20(new.salary, new.meal_allowance, new.position_allowance, new.fixed_fuel_allowance, new.fixed_performance_bonus, new.other_fixed_allowance);
  dep_count := least(greatest(coalesce(new.health_dependents_count,0),0),3);

  -- V20：投保薪資包含固定經常性給與，不只本薪。
  new.labor_insured_salary := peihong_labor_grade_v20(wage_base);
  new.health_insured_salary := peihong_health_grade_v20(wage_base);
  new.labor_insurance := round(new.labor_insured_salary * 0.125 * 0.2);
  new.health_insurance := round(new.health_insured_salary * 0.0517 * 0.3);
  new.dependent_health_insurance := round(new.health_insured_salary * 0.0517 * 0.3 * dep_count);
  new.health_dependents_count := dep_count;

  if coalesce(new.annual_leave_total,0) = 0 and new.hire_date is not null then
    new.annual_leave_total := peihong_annual_leave_days_v20(new.hire_date, current_date);
  end if;
  new.annual_leave_used := coalesce(new.annual_leave_used,0);
  new.annual_leave_remaining := greatest(coalesce(new.annual_leave_total,0) - coalesce(new.annual_leave_used,0), 0);

  return new;
end; $$;

drop trigger if exists trg_peihong_sync_employee_insurance on employees;

create trigger trg_peihong_sync_employee_v20
before insert or update
on employees
for each row
execute function peihong_sync_employee_v20();

  annual_leave_used = coalesce(annual_leave_used,0),
  annual_leave_remaining = greatest(coalesce(annual_leave_total,0) - coalesce(annual_leave_used,0),0);
