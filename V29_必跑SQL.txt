begin
  if coalesce(p_salary,0) >= 43901 then return 45800;
  elsif coalesce(p_salary,0) >= 42001 then return 43900;
  elsif coalesce(p_salary,0) >= 40101 then return 42000;
  elsif coalesce(p_salary,0) >= 38201 then return 40100;
  elsif coalesce(p_salary,0) >= 36301 then return 38200;
  elsif coalesce(p_salary,0) >= 34801 then return 36300;

  elsif coalesce(p_salary,0) >= 29501 then return 30300;
  else return 29500;
  end if;
end;
$$;

create or replace function peihong_sync_employee_v29()
returns trigger language plpgsql as $$
declare
  dep_count numeric;
  manual_labor numeric;
  manual_health numeric;
begin
  dep_count := least(greatest(coalesce(new.health_dependents_count,0),0),3);
  manual_labor := coalesce(nullif(new.manual_labor_insured_salary,0), nullif(new.labor_insured_salary,0), 0);
  manual_health := coalesce(nullif(new.manual_health_insured_salary,0), nullif(new.health_insured_salary,0), 0);

  if coalesce(new.insurance_manual_override,false) = true then
    new.manual_labor_insured_salary := manual_labor;
    new.manual_health_insured_salary := manual_health;
    new.labor_insured_salary := manual_labor;
    new.health_insured_salary := manual_health;
  else
    new.manual_labor_insured_salary := 0;
    new.manual_health_insured_salary := 0;
    new.labor_insured_salary := peihong_labor_grade_v29(new.salary);
    new.health_insured_salary := peihong_health_grade_v29(new.salary);
  end if;

  new.labor_insurance := round(coalesce(new.labor_insured_salary,0) * 0.125 * 0.2);
  new.health_insurance := round(coalesce(new.health_insured_salary,0) * 0.0517 * 0.3);
  new.dependent_health_insurance := round(coalesce(new.health_insured_salary,0) * 0.0517 * 0.3 * dep_count);
  new.health_dependents_count := dep_count;

  new.annual_leave_used := coalesce(new.annual_leave_used,0);
  new.annual_leave_total := coalesce(new.annual_leave_total,0);
  new.annual_leave_remaining := greatest(new.annual_leave_total - new.annual_leave_used, 0);

  return new;
end;
$$;

create or replace function peihong_round_overtime_v29(p_hours numeric)
returns numeric language sql immutable as $$
  select case
    when coalesce(p_hours,0) < 0.5 then floor(coalesce(p_hours,0))
    when (coalesce(p_hours,0) - floor(coalesce(p_hours,0))) >= 0.5 then ceil(coalesce(p_hours,0))
    else floor(coalesce(p_hours,0))
  end;
$$;

create or replace function peihong_overtime_hours_after_17_v29(p_start time, p_end time)
returns numeric language plpgsql immutable as $$
declare
  s interval;
  e interval;
  base interval := interval '17 hours';
  diff_hours numeric;
begin
  if p_start is null or p_end is null then return 0; end if;
  s := p_start - time '00:00';
  e := p_end - time '00:00';
  if s < base then s := base; end if;
  if e <= s then return 0; end if;
  diff_hours := extract(epoch from (e - s)) / 3600.0;
  return peihong_round_overtime_v29(diff_hours);
end;
$$;

create or replace function peihong_sync_overtime_breakdown_v29()
returns trigger language plpgsql as $$
declare
  hrs numeric;
begin
  hrs := peihong_overtime_hours_after_17_v29(coalesce(new.start_time, new.overtime_start), coalesce(new.end_time, new.overtime_end));

  new.overtime_hours := hrs;
  new.total_hours := hrs;
  new.calculated_hours := hrs;
  new.day_type := coalesce(new.day_type, new.overtime_day_type, 'weekday');
  new.overtime_day_type := coalesce(new.overtime_day_type, new.day_type, 'weekday');

  new.weekday_first2_hours := 0;
  new.weekday_after2_hours := 0;
  new.restday_first2_hours := 0;
  new.restday_3to8_hours := 0;
  new.restday_9to12_hours := 0;
  new.holiday_hours := 0;
  new.regular_holiday_hours := 0;

  if new.day_type = 'weekday' then
    new.weekday_first2_hours := least(hrs, 2);
    new.weekday_after2_hours := greatest(hrs - 2, 0);
  elsif new.day_type = 'restday' or new.day_type = 'rest_day' then
    new.restday_first2_hours := least(hrs, 2);
    new.restday_3to8_hours := least(greatest(hrs - 2, 0), 6);
    new.restday_9to12_hours := greatest(hrs - 8, 0);
  elsif new.day_type = 'holiday' or new.day_type = 'national_holiday' then
    new.holiday_hours := hrs;
  elsif new.day_type = 'regular_holiday' then
    new.regular_holiday_hours := hrs;
  end if;

  new.overtime_breakdown := jsonb_build_object(
    'day_type', new.day_type,
    'total_hours', hrs,
    'weekday_first2_hours', new.weekday_first2_hours,
    'weekday_after2_hours', new.weekday_after2_hours,
    'restday_first2_hours', new.restday_first2_hours,
    'restday_3to8_hours', new.restday_3to8_hours,
    'restday_9to12_hours', new.restday_9to12_hours,
    'holiday_hours', new.holiday_hours,
    'regular_holiday_hours', new.regular_holiday_hours
  );

  return new;
end;
$$;
