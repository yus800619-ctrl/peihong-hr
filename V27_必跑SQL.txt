-- 進位規則：以0.5小時為單位向上進位
create or replace function peihong_round_up_half_hour_v27(p_hours numeric)
returns numeric
language sql
immutable
as $$
  select greatest(ceil(coalesce(p_hours,0) * 2) / 2, 0);
$$;

create or replace function peihong_overtime_hours_after_17_v27(p_start time, p_end time)
returns numeric
language plpgsql
immutable
as $$
declare
  s interval;
  e interval;
  base interval := interval '17 hours';
  diff_hours numeric;
begin
  if p_start is null or p_end is null then return 0; end if;
  s := p_start - time '00:00';
  e := p_end - time '00:00';
  if s < base then s := base; end if;
  if e <= s then return 0; end if;
  diff_hours := extract(epoch from (e - s)) / 3600.0;
  return peihong_round_up_half_hour_v27(diff_hours);
end;
$$;

create or replace function peihong_sync_overtime_breakdown_v27()
returns trigger
language plpgsql
as $$
declare
  hrs numeric;
  d_type text;
begin
  hrs := peihong_overtime_hours_after_17_v27(coalesce(new.start_time,new.overtime_start), coalesce(new.end_time,new.overtime_end));
  d_type := coalesce(new.overtime_day_type, new.day_type, 'weekday');

  new.overtime_hours := hrs;
  new.total_hours := hrs;
  new.calculated_hours := hrs;
  new.overtime_day_type := d_type;
  new.day_type := d_type;
  new.start_time := coalesce(new.start_time,new.overtime_start);
  new.end_time := coalesce(new.end_time,new.overtime_end);
  new.overtime_start := coalesce(new.overtime_start,new.start_time);
  new.overtime_end := coalesce(new.overtime_end,new.end_time);

  new.weekday_first2_hours := 0;
  new.weekday_after2_hours := 0;
  new.restday_first2_hours := 0;
  new.restday_3to8_hours := 0;
  new.restday_9to12_hours := 0;
  new.holiday_hours := 0;
  new.regular_holiday_hours := 0;

  if d_type = 'weekday' then
    new.weekday_first2_hours := least(hrs, 2);
    new.weekday_after2_hours := greatest(hrs - 2, 0);
    new.overtime_breakdown := jsonb_build_object('weekday_first2', new.weekday_first2_hours, 'weekday_after2', new.weekday_after2_hours);
  elsif d_type = 'rest_day' or d_type = 'restday' then
    new.restday_first2_hours := least(hrs, 2);
    new.restday_3to8_hours := least(greatest(hrs - 2, 0), 6);
    new.restday_9to12_hours := greatest(hrs - 8, 0);
    new.overtime_breakdown := jsonb_build_object('rest_first2', new.restday_first2_hours, 'rest_3to8', new.restday_3to8_hours, 'rest_9to12', new.restday_9to12_hours);
  elsif d_type = 'national_holiday' or d_type = 'holiday' then
    new.holiday_hours := hrs;
    new.overtime_breakdown := jsonb_build_object('holiday_within8', least(hrs,8), 'holiday_after8_first2', least(greatest(hrs-8,0),2), 'holiday_after8_after2', greatest(hrs-10,0));
  elsif d_type = 'regular_holiday' then
    new.regular_holiday_hours := hrs;
    new.overtime_breakdown := jsonb_build_object('regular_emergency', least(hrs,8), 'regular_after8_first2', least(greatest(hrs-8,0),2), 'regular_after8_after2', greatest(hrs-10,0));
  end if;

  return new;
end;
$$;

drop trigger if exists trg_peihong_sync_overtime_breakdown_v26 on overtime_requests;
drop trigger if exists trg_peihong_sync_overtime_breakdown_v27 on overtime_requests;

create trigger trg_peihong_sync_overtime_breakdown_v27
before insert or update of start_time, end_time, overtime_start, overtime_end, overtime_day_type, day_type
on overtime_requests
for each row
execute function peihong_sync_overtime_breakdown_v27();
