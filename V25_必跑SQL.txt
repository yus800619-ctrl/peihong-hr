create or replace function peihong_labor_grade_v25(p_salary numeric)
returns numeric language plpgsql immutable as $$
begin
  if coalesce(p_salary,0) >= 43901 then return 45800;
  elsif coalesce(p_salary,0) >= 42001 then return 43900;
  elsif coalesce(p_salary,0) >= 40101 then return 42000;
  elsif coalesce(p_salary,0) >= 38201 then return 40100;
  elsif coalesce(p_salary,0) >= 36301 then return 38200;
  elsif coalesce(p_salary,0) >= 34801 then return 36300;
  elsif coalesce(p_salary,0) >= 33301 then return 34800;
  elsif coalesce(p_salary,0) >= 31801 then return 33300;
  elsif coalesce(p_salary,0) >= 30301 then return 31800;
  elsif coalesce(p_salary,0) >= 29501 then return 30300;
  else return 29500;
  end if;
end;
$$;

create or replace function peihong_health_grade_v25(p_salary numeric)

begin
  if coalesce(p_salary,0) >= 101101 then return 105600;
  elsif coalesce(p_salary,0) >= 96601 then return 101100;
  elsif coalesce(p_salary,0) >= 92101 then return 96600;
  elsif coalesce(p_salary,0) >= 87601 then return 92100;
  elsif coalesce(p_salary,0) >= 83901 then return 87600;
  elsif coalesce(p_salary,0) >= 80201 then return 83900;
  elsif coalesce(p_salary,0) >= 76501 then return 80200;
  elsif coalesce(p_salary,0) >= 72801 then return 76500;
  elsif coalesce(p_salary,0) >= 69801 then return 72800;
  elsif coalesce(p_salary,0) >= 66801 then return 69800;
  elsif coalesce(p_salary,0) >= 63801 then return 66800;
  elsif coalesce(p_salary,0) >= 60801 then return 63800;
  elsif coalesce(p_salary,0) >= 57801 then return 60800;
  elsif coalesce(p_salary,0) >= 55401 then return 57800;
  elsif coalesce(p_salary,0) >= 53001 then return 55400;
  elsif coalesce(p_salary,0) >= 50601 then return 53000;
  elsif coalesce(p_salary,0) >= 48201 then return 50600;
  elsif coalesce(p_salary,0) >= 45801 then return 48200;
  elsif coalesce(p_salary,0) >= 43901 then return 45800;

create or replace function peihong_sync_employee_v25()
returns trigger language plpgsql as $$
declare
  dep_count numeric;
  labor_grade numeric;
  health_grade numeric;
begin
  dep_count := least(greatest(coalesce(new.health_dependents_count,0),0),3);

  if coalesce(new.insurance_manual_override,false) then
    labor_grade := case when coalesce(new.labor_insured_salary,0) > 0 then new.labor_insured_salary else peihong_labor_grade_v25(new.salary) end;
    health_grade := case when coalesce(new.health_insured_salary,0) > 0 then new.health_insured_salary else peihong_health_grade_v25(new.salary) end;
  else
    labor_grade := peihong_labor_grade_v25(new.salary);
    health_grade := peihong_health_grade_v25(new.salary);
  end if;

  new.labor_insured_salary := labor_grade;
  new.health_insured_salary := health_grade;
  new.labor_insurance := round(labor_grade * 0.125 * 0.2);
  new.health_insurance := round(health_grade * 0.0517 * 0.3);
  new.dependent_health_insurance := round(health_grade * 0.0517 * 0.3 * dep_count);
  new.health_dependents_count := dep_count;

  return new;
end;
$$;

drop trigger if exists trg_peihong_sync_employee_v22 on employees;
drop trigger if exists trg_peihong_sync_employee_v21 on employees;
drop trigger if exists trg_peihong_sync_employee_v20 on employees;
drop trigger if exists trg_peihong_sync_employee_v25 on employees;

create trigger trg_peihong_sync_employee_v25
before insert or update
on employees
for each row
execute function peihong_sync_employee_v25();

update employees
set insurance_manual_override = coalesce(insurance_manual_override,false),
    health_dependents_count = coalesce(health_dependents_count,0);

grant usage on schema public to anon, authenticated;
grant select, insert, update, delete on employees to anon, authenticated;
grant select, insert, update, delete on overtime_requests to anon, authenticated;
grant select, insert, update, delete on leave_requests to anon, authenticated;
notify pgrst, 'reload schema';

select employee_no, name, salary, insurance_manual_override, labor_insured_salary, health_insured_salary, labor_insurance, health_insurance, dependent_health_insurance
from employees
order by employee_no;
